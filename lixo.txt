import { useState, useEffect } from "react";
import Header from "../components/ui/Header";
import RenderizacaoUsuario from "../components/usuario/renderUsuario";
import ModalUsuario from "../components/usuario/modalUsuario";

export default function UserProfileManager() {
  const [funcionarios, setFuncionarios] = useState([]);
  const [id, setId] = useState("");

  const buscarFuncionarios = async () => {
    try {
      const resposta = await fetch("http://localhost:1050/funcionarios");
      const dados = await resposta.json();
      setFuncionarios(dados);

      // Seleciona o primeiro usuário por padrão caso nenhum esteja selecionado
      if (dados.length > 0 && !id) {
        setId(dados[0].id_funcionario);
      }
    } catch (error) {
      console.error("Erro ao buscar funcionários:", error);
    }
  };

  useEffect(() => {
    buscarFuncionarios();
  }, []);

  const atualizarFuncionarioNaLista = (funcionarioAtualizado) => {
    setFuncionarios((listaAnterior) =>
      listaAnterior.map((func) =>
        func.id_funcionario === funcionarioAtualizado.id_funcionario
          ? { ...func, ...funcionarioAtualizado }
          : func,
      ),
    );
  };

  return (
    <>
      <Header
        title="Usuarios"
        subtitle="Veja seu perfil, navegue pelos usuários e cadastre novos acessos."
      />

      <main>
        <section className="flex items-center gap-36">
          <div className="w-96 ml-20 border border-slate-200 rounded-xl overflow-hidden shadow-sm bg-white mt-10">
            <RenderizacaoUsuario
              funcionarios={funcionarios}
              idSelecionado={id}
              setId={setId}
            />
          </div>

          <div>
            <ModalUsuario
              id={id}
              funcionarios={funcionarios}
              aoAtualizarFuncionario={atualizarFuncionarioNaLista}
            />
          </div>
        </section>
      </main>
    </>
  );
}
